cmake_minimum_required(VERSION 3.16)
project(flash_fr3_model_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Franka REQUIRED)
find_package(franka_semantic_components REQUIRED)
add_library(${PROJECT_NAME} SHARED src/model_controller.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} controller_interface controller_manager_msgs hardware_interface pluginlib
  rclcpp rclcpp_lifecycle realtime_tools std_msgs)
target_link_libraries(${PROJECT_NAME} Franka::Franka)
add_library(flash_fr3_stream_plugin SHARED src/stream_controller.cpp)
target_include_directories(flash_fr3_stream_plugin PUBLIC include)
ament_target_dependencies(flash_fr3_stream_plugin controller_interface controller_manager_msgs
  hardware_interface pluginlib rclcpp rclcpp_lifecycle realtime_tools std_msgs)
target_link_libraries(flash_fr3_stream_plugin Franka::Franka)
add_library(flash_fr3_task_plugin SHARED src/task_controller.cpp)
target_include_directories(flash_fr3_task_plugin PUBLIC include)
ament_target_dependencies(flash_fr3_task_plugin controller_interface controller_manager_msgs
  hardware_interface pluginlib rclcpp rclcpp_lifecycle realtime_tools std_msgs)
target_link_libraries(flash_fr3_task_plugin Franka::Franka)
add_library(flash_fr3_torque_plugin SHARED src/torque_controller.cpp)
target_include_directories(flash_fr3_torque_plugin PUBLIC include)
ament_target_dependencies(flash_fr3_torque_plugin controller_interface controller_manager_msgs
  hardware_interface pluginlib rclcpp rclcpp_lifecycle realtime_tools std_msgs franka_semantic_components)
target_link_libraries(flash_fr3_torque_plugin Franka::Franka)
add_library(flash_fr3_continuous_effort_plugin SHARED src/continuous_effort_controller.cpp)
target_include_directories(flash_fr3_continuous_effort_plugin PUBLIC include)
ament_target_dependencies(flash_fr3_continuous_effort_plugin controller_interface controller_manager_msgs
  hardware_interface pluginlib rclcpp rclcpp_lifecycle realtime_tools std_msgs franka_semantic_components)
target_link_libraries(flash_fr3_continuous_effort_plugin Franka::Franka)
install(TARGETS flash_fr3_continuous_effort_plugin LIBRARY DESTINATION lib COMPONENT continuous_effort)
pluginlib_export_plugin_description_file(controller_interface model_plugins.xml)
add_executable(preview_model src/preview_model.cpp)
target_include_directories(preview_model PRIVATE include)
install(TARGETS ${PROJECT_NAME} flash_fr3_stream_plugin LIBRARY DESTINATION lib)
install(TARGETS preview_model RUNTIME DESTINATION lib/${PROJECT_NAME})
add_executable(preview_task src/preview_task.cpp)
target_include_directories(preview_task PRIVATE include)
install(TARGETS flash_fr3_task_plugin LIBRARY DESTINATION lib)
install(TARGETS flash_fr3_torque_plugin LIBRARY DESTINATION lib)
install(TARGETS preview_task RUNTIME DESTINATION lib/${PROJECT_NAME})
add_executable(preview_continuous src/preview_continuous.cpp)
target_include_directories(preview_continuous PRIVATE include)
install(TARGETS preview_continuous RUNTIME DESTINATION lib/${PROJECT_NAME})
add_executable(preview_continuous_effort src/preview_continuous_effort.cpp)
target_include_directories(preview_continuous_effort PRIVATE include)
install(TARGETS preview_continuous_effort RUNTIME DESTINATION lib/${PROJECT_NAME} COMPONENT continuous_effort)
install(DIRECTORY include/ DESTINATION include)
