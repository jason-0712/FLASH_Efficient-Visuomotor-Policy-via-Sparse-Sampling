cmake_minimum_required(VERSION 3.16)
project(flash_fr3_pilot_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Franka REQUIRED)
add_library(${PROJECT_NAME} SHARED src/pilot_controller.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} controller_interface controller_manager_msgs hardware_interface pluginlib
  rclcpp rclcpp_lifecycle realtime_tools std_msgs)
target_link_libraries(${PROJECT_NAME} Franka::Franka)
pluginlib_export_plugin_description_file(controller_interface pilot_plugins.xml)
add_executable(preview_pilot src/preview_pilot.cpp)
target_include_directories(preview_pilot PRIVATE include)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(TARGETS preview_pilot RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_pilot_core tests/test_pilot_core.cpp)
  target_include_directories(test_pilot_core PRIVATE include)
  ament_add_gtest(test_pilot_native_stop tests/test_pilot_native_stop.cpp)
  target_include_directories(test_pilot_native_stop PRIVATE include)
  target_link_libraries(test_pilot_native_stop ${PROJECT_NAME} Franka::Franka)
  ament_target_dependencies(test_pilot_native_stop controller_interface controller_manager_msgs
    hardware_interface rclcpp rclcpp_lifecycle realtime_tools std_msgs)
endif()
ament_package()
